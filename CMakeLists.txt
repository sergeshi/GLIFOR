cmake_minimum_required(VERSION 3.21)
project(GLIFOR VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
qt_add_executable(GLIFOR src/main.cpp src/glyph.h src/glyph.cpp src/glyphcanvas.h src/glyphcanvas.cpp src/mainwindow.h src/mainwindow.cpp)
target_link_libraries(GLIFOR PRIVATE Qt6::Widgets)
if(MSVC)
  target_compile_options(GLIFOR PRIVATE /W4 /permissive-)
else()
  target_compile_options(GLIFOR PRIVATE -Wall -Wextra -Wpedantic)
endif()
